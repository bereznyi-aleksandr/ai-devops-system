# НАЗНАЧЕНИЕ: аудит ChatGPT S6-prep R7, M-S6-R7-01 — оператор вставляет у провайдера только тот user-data.yaml,
#   который проверил независимый аудитор. Функция читает файл ОДИН раз, показывает вторую строку (отпечаток ключа
#   подписи) и просит ввести первые 16 знаков USERDATA_SHA256 из ответа аудитора. Сумму файла функция НЕ печатает:
#   её нужно взять из ответа аудитора, а не с экрана этой функции. Совпало — в буфер обмена идут ровно прочитанные
#   байты; не совпало — ничего не копируется. Изменение файла после аудита (лишний ключ, sudo, служба) меняет сумму.
# КАК ПОЛЬЗОВАТЬСЯ: текст функции берётся с GitHub по закреплённому коммиту (OD-5 v1.8, шаг 7), не из локальной копии.
#   Файл назван .txt, чтобы его нельзя было запустить как сценарий.
# ОГРАНИЧЕНИЯ: остаточный риск — подмена буфера обмена или ввода процессом той же учётной записи Windows; его
#   закрывает сверка ключа на сервере (OD-5 шаг 8а) только для ключа подписи.
function Copy-ZavodUserData {
  param([Parameter(Mandatory = $true)][string]$Path)
  $b = [System.IO.File]::ReadAllBytes($Path)
  $h = -join ([System.Security.Cryptography.SHA256]::Create().ComputeHash($b) | ForEach-Object { $_.ToString('x2') })
  $text = [System.Text.Encoding]::UTF8.GetString($b)
  $lines = $text -split "`n"
  if ($lines.Count -lt 3 -or $lines[0] -ne '#cloud-config') { throw 'Это не файл user-data. Ничего не скопировано.' }
  Write-Host $lines[1]
  $ans = Read-Host 'Введите первые 16 знаков USERDATA_SHA256 из ответа аудитора'
  if ($null -eq $ans -or $ans.Trim().ToLowerInvariant() -ne $h.Substring(0, 16)) { throw 'Код не совпал. Ничего не скопировано.' }
  Set-Clipboard -Value $text
  Write-Host 'Скопировано в буфер обмена. Вставьте в поле initialization scripts.'
}
